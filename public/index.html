// =================================================== // MAMANGGS - BACKEND SERVERLESS API HANDLER // =================================================== const express = require('express'); const cors = require('cors'); const app = express(); app.use(cors()); app.use(express.json()); // Endpoint Tes Status API app.get('/api', (req, res) => { res.status(200).json({ status: 'success', message: 'MamangGS API Service is running smoothly 🚀', timestamp: new Date().toISOString() }); }); // Endpoint Cek Transaksi / Invoice Sederhana app.get('/api/check-order/:invoice', async (req, res) => { const { invoice } = req.params; if (!invoice) { return res.status(400).json({ success: false, message: 'Nomor invoice wajib diisi' }); } // Respon placeholder sebelum terhubung penuh ke Supabase/Payment Gateway res.status(200).json({ success: true, data: { invoice: invoice, status: 'PENDING', message: 'Menunggu pembayaran via QRIS' } }); }); module.exports = app;
